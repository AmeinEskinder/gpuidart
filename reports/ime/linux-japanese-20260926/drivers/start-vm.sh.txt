#!/bin/sh
set -eu
cd /work
exec > /work/qemu.stdout.log 2> /work/qemu.stderr.log
exec qemu-system-x86_64 \
  -enable-kvm -cpu host -smp 2 -m 4096 \
  -drive file=guest.qcow2,if=virtio,format=qcow2 \
  -drive file=seed.img,if=virtio,format=raw,readonly=on \
  -device virtio-net-pci,netdev=network \
  -netdev user,id=network,hostfwd=tcp:127.0.0.1:22229-:22 \
  -vga virtio -display none \
  -serial file:/work/serial.log \
  -qmp unix:/work/qmp.sock,server=on,wait=off \
  -pidfile /work/qemu.pid
