import json
import os
from pathlib import Path
import subprocess
import time

root = Path('/home/evaluator')
for line in (root / 'session.env').read_text().splitlines():
    key, _, value = line.partition('=')
    if key in {'DISPLAY', 'DBUS_SESSION_BUS_ADDRESS', 'XMODIFIERS', 'LANG', 'LC_ALL'}:
        os.environ[key] = value
path = root / '.config/fcitx5/conf/xim.conf'
before = path.read_text() if path.exists() else None
if before is not None:
    raise SystemExit('Retain and inspect the existing XIM configuration before changing it')
path.parent.mkdir(parents=True, exist_ok=True)
path.write_text('UseOnTheSpot=True\n')
log = (root / 'evidence/fcitx-on-the-spot.log').open('w')
process = subprocess.Popen(['fcitx5', '--replace', '--disable=wayland'], stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
time.sleep(2)
if process.poll() is not None:
    raise SystemExit('Fcitx replacement exited')
report = {'before': before, 'after': path.read_text(), 'restarted_fcitx_pid': process.pid, 'time_unix': time.time(), 'scope': 'Only the isolated guest input method configuration changed; the SDK package is unchanged.'}
(root / 'evidence/xim-configuration-change.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps(report))
