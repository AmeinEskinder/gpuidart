import datetime
import hashlib
import json
from pathlib import Path
import shutil

cache = Path('.cache/linux-desktop')
source = cache / 'evidence-final-1'
target = Path('reports/ime/linux-japanese-20260926')
target.mkdir(exist_ok=True)

def copy(src, dst):
    dst.parent.mkdir(parents=True, exist_ok=True)
    if dst.exists() and dst.read_bytes() != src.read_bytes():
        raise SystemExit(f'Refusing to overwrite different evidence: {dst}')
    shutil.copy2(src, dst)

for src in source.rglob('*'):
    if not src.is_file(): continue
    relative = src.relative_to(source)
    if src.suffix == '.png': dst = target / 'screenshots' / src.name
    elif src.stem.startswith(('aot-', 'jit-sequence-2-')) and src.suffix == '.json' and src.stem != 'aot-close-and-phase':
        dst = target / 'captures' / src.name
    elif len(relative.parts) > 1: dst = target / relative
    else: dst = target / 'raw' / src.name
    copy(src, dst)
for name in ['fcitx.log', 'openbox.log']:
    copy(cache / 'guest-evidence-initial' / name, target / 'raw/initial-session' / name)
for name in ['build.log', 'build-host-ipv4.log', 'apt-probe.log', 'cloud-init-wait.log', 'clean-verification.log', 'clean-verification2.log', 'remove-git.log', 'gdb-install.log', 'openbox-backtrace.log', 'startup-backtrace.log', 'qxl-unavailable.stderr.log', 'development-phase.log', 'development-phase-attempt2.log', 'jit-sequence-1.log', 'jit-sequence-2.log']:
    copy(cache / name, target / 'raw' / name)
for name in ['Dockerfile', 'provision.py', 'start-vm.sh', 'verify-clean.sh', 'verify-clean-v2.sh', 'configure_xim.py', 'probe_startup.py', 'restart_desktop.py', 'startup.gdb', 'advance_guest.py', 'guest.py', 'guest-env.sh', 'observe.py', 'run_action.py', 'reload_sequence.dart', 'run_jit.sh', 'copy_sdk.py', 'collect_evidence.py']:
    copy(cache / name, target / 'drivers' / (name + '.txt'))
copy(cache / 'image.json', target / 'image.json')
records = []
for screenshot in sorted((target / 'screenshots').glob('*.png')):
    metadata = target / 'captures' / (screenshot.stem + '.json')
    record = json.loads(metadata.read_text(encoding='utf-8'))
    records.append({
        'path': screenshot.relative_to(target).as_posix(),
        'sha256': hashlib.file_digest(screenshot.open('rb'), 'sha256').hexdigest(),
        'bytes': screenshot.stat().st_size,
        'captured_utc': datetime.datetime.fromtimestamp(record['time_unix'], datetime.timezone.utc).isoformat(),
        'metadata': metadata.relative_to(target).as_posix(),
        'metadata_sha256': hashlib.file_digest(metadata.open('rb'), 'sha256').hexdigest(),
    })
(target / 'capture-manifest.json').write_text(json.dumps({'observer': 'agent', 'screenshots': records}, indent=2) + '\n', encoding='utf-8')
verification = json.loads((source / 'clean-attempt2/verification.json').read_text(encoding='utf-8'))
